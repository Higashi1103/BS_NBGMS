DROP TABLE IF EXISTS bs_nbgms2025.kurasu_info;

CREATE TABLE bs_nbgms2025.kurasu_info (
  KURASU_ID INT NOT NULL AUTO_INCREMENT,
  KURASU_NAMAE VARCHAR(30) NOT NULL,
  KURASU_KAMOKU VARCHAR(50) NOT NULL,
  TANTO_SENSEI VARCHAR(20) NOT NULL,
  DELETE_FLAG INT NOT NULL DEFAULT 0,
  MEMO VARCHAR(100),
  TOUROKUBI DATE NOT NULL DEFAULT (CURRENT_DATE),
  KOUSHINBI DATE,
  SAKUSEISHA VARCHAR(20),
  YOBI_1 VARCHAR(200),
  YOBI_2 VARCHAR(200),
  YOBI_3 VARCHAR(200),
  PRIMARY KEY (KURASU_ID)
);
